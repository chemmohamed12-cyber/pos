/*
 * Org-engine migration up / down / up check on ONE disposable database (cluster 5546, name daii_test_biz_t96_migdown).
 * The DB guard (db-guard-register) refuses anything but an approved disposable name; the script additionally refuses a database that already exists.
 * Applies EVERY migration of the compiled tree from an empty database, then no-op re-run, then seeds rows that use the new columns, undoes the 7 org-engine migrations
 * (newest first), records what the rollback actually did to schema and data, re-applies, and compares schema fingerprints.
 */
require('./db-guard-register');
import { execSync } from 'child_process';
import { createHash } from 'crypto';
import { randomUUID } from 'crypto';
import { DataSource } from 'typeorm';
import { createMigrationDataSource, discoverMigrationTags } from '../../src/foundation/release/migration-datasource';

const NEW = ['OrgUnitLifecycle1790021000000', 'AddVehicleOrgUnit1790021100000', 'AddSalesOrderBranch1790021200000', 'AddSalesInvoiceBranch1790021300000', 'AddFulfillmentOrderBranch1790021400000', 'AddShipmentBranch1790021500000', 'OrgMigrationLog1790021600000'];
const base = { host: process.env.DB_HOST!, port: Number(process.env.DB_PORT), username: process.env.DB_USER!, password: process.env.DB_PASSWORD! };
const DB = process.env.MIG_DB!;
const APP_ROOT = process.cwd();
const TABLES = ['org_units', 'org_memberships', 'access_scopes', 'org_migration_log', 'vehicles', 'sales_orders', 'sales_invoices', 'fulfillment_orders', 'shipments', 'warehouses', 'purchase_orders', 'treasury_accounts'];

async function fingerprint(ds: DataSource) {
  const cols = await ds.query(`SELECT table_name,column_name,data_type,is_nullable,column_default FROM information_schema.columns WHERE table_schema=current_schema() AND table_name = ANY($1) ORDER BY 1,2`, [TABLES]);
  const idx = await ds.query(`SELECT tablename,indexname,indexdef FROM pg_indexes WHERE schemaname=current_schema() AND tablename = ANY($1) ORDER BY 1,2`, [TABLES]);
  const con = await ds.query(`SELECT conrelid::regclass::text AS tbl, conname, contype, confdeltype, pg_get_constraintdef(oid) AS def FROM pg_constraint WHERE connamespace = current_schema()::regnamespace AND conrelid::regclass::text = ANY($1) ORDER BY 1,2`, [TABLES]);
  const hash = createHash('sha256').update(JSON.stringify({ cols, idx, con })).digest('hex');
  return { hash, columns: cols.length, indexes: idx.length, constraints: con.length, cols, idx, con };
}
const brief = (fp: Awaited<ReturnType<typeof fingerprint>>) => ({ hash: fp.hash.slice(0, 16), columns: fp.columns, indexes: fp.indexes, constraints: fp.constraints });

async function main() {
  const out: Record<string, unknown> = {};
  out.testedCommit = execSync('git rev-parse HEAD', { cwd: APP_ROOT }).toString().trim();
  out.workingTreeDirtyFiles = execSync('git status --short', { cwd: APP_ROOT }).toString().split('\n').filter((l) => l.trim() && !l.includes('_tmp-migration-updown.ts')).length;
  out.startedAt = new Date().toISOString();
  const admin = new DataSource({ type: 'postgres', ...base, database: 'postgres' });
  await admin.initialize();
  out.cluster = (await admin.query(`SELECT inet_server_port() AS port, version() AS version, (SELECT system_identifier FROM pg_control_system()) AS system_identifier`))[0];
  const before = (await admin.query(`SELECT datname FROM pg_database WHERE datname NOT LIKE 'template%' ORDER BY 1`)).map((r: { datname: string }) => r.datname);
  if (before.includes(DB)) throw new Error(`refusing: ${DB} already exists - disposable databases must be fresh`);
  await admin.query(`CREATE DATABASE "${DB}"`);
  out.disposableDatabase = { name: DB, existedBefore: false, createdByThisRun: true, nameMatchesGuardPattern: /^daii_test_biz_t[0-9]{2}[a-z0-9_]*$/.test(DB), otherDatabasesOnClusterUntouched: before.filter((d: string) => d !== DB).length };

  const tags = discoverMigrationTags(APP_ROOT);
  out.tags = tags;
  const ds = createMigrationDataSource(APP_ROOT, { ...base, database: DB } as never, tags);
  await ds.initialize();
  out.connectedTo = (await ds.query(`SELECT current_database() AS db, inet_server_port() AS port`))[0];

  // 1. up from empty
  const ran = await ds.runMigrations({ transaction: 'each' });
  const fp1 = await fingerprint(ds);
  out.step1_up = { migrationsApplied: ran.length, orgEngineMigrationsApplied: ran.filter((m) => NEW.includes(m.name)).map((m) => m.name), executedRows: Number((await ds.query(`SELECT count(*)::int n FROM migrations`))[0].n), schema: brief(fp1) };
  // 2. up again = no-op
  out.step2_upAgainApplied = (await ds.runMigrations({ transaction: 'each' })).length;

  // 3. seed rows that USE the new columns
  const userId = randomUUID();
  await ds.query(`INSERT INTO users (id,"fullName",email) VALUES ($1,'Mig Tester','mig-tester@test.local')`, [userId]);
  const rootId = randomUUID(), branchId = randomUUID(), archivedId = randomUUID(), cust = randomUUID(), so = randomUUID(), veh = randomUUID();
  await ds.query(`INSERT INTO org_units (id,type,name) VALUES ($1,'company','Root')`, [rootId]);
  await ds.query(`INSERT INTO org_units (id,type,name,"parentId") VALUES ($1,'branch','Branch',$2)`, [branchId, rootId]);
  await ds.query(`INSERT INTO org_units (id,type,name,"parentId",status,"archivedAt") VALUES ($1,'branch','Old branch',$2,'archived',now())`, [archivedId, rootId]);
  await ds.query(`INSERT INTO customers (id,code,name,"creditLimit","paymentTermsDays") VALUES ($1,'C-1','Cust','1000',30)`, [cust]);
  await ds.query(`INSERT INTO sales_orders (id,"orderNumber","customerId","salesRepUserId","totalAmount","repSequenceNumber","createdByUserId","branchOrgUnitId") VALUES ($1,'SO-MIG',$2,$3,'250.00',1,$3,$4)`, [so, cust, userId, branchId]);
  await ds.query(`INSERT INTO vehicles (id,code,"plateNumber",branch,status,"orgUnitId") VALUES ($1,'V-MIG','P 1','Branch','active',$2)`, [veh, branchId]);
  await ds.query(`INSERT INTO org_migration_log ("runId","tableName","columnName","recordId","toValue","source") VALUES ($1,'sales_orders','branchOrgUnitId',$2,$3,'warehouse.owning_branch')`, [randomUUID(), so, branchId]);
  const seeded = async () => ({
    orgUnits: Number((await ds.query(`SELECT count(*)::int n FROM org_units`))[0].n),
    salesOrder: (await ds.query(`SELECT "orderNumber","totalAmount"::text AS total FROM sales_orders WHERE id=$1`, [so]))[0] ?? null,
    vehicle: (await ds.query(`SELECT code,"plateNumber",branch FROM vehicles WHERE id=$1`, [veh]))[0] ?? null,
  });
  out.step3_seededBeforeDown = { ...(await seeded()), branchStampOnOrder: (await ds.query(`SELECT "branchOrgUnitId" FROM sales_orders WHERE id=$1`, [so]))[0].branchOrgUnitId === branchId, vehicleOrgUnit: (await ds.query(`SELECT "orgUnitId" FROM vehicles WHERE id=$1`, [veh]))[0].orgUnitId === branchId, logRows: 1, archivedUnitStatus: 'archived' };

  // 4. down: the 7 org-engine migrations, newest first
  const undoneNames: string[] = [];
  for (let i = 0; i < NEW.length; i++) {
    const last = (await ds.query(`SELECT name FROM migrations ORDER BY id DESC LIMIT 1`))[0].name as string;
    await ds.undoLastMigration({ transaction: 'each' });
    undoneNames.push(last);
  }
  const fp2 = await fingerprint(ds);
  const has = async (sql: string) => (await ds.query(sql)).length > 0;
  out.step4_down = {
    undone: undoneNames,
    undoneExactlyTheSevenOrgEngineMigrations: JSON.stringify([...undoneNames].sort()) === JSON.stringify([...NEW].sort()),
    executedRows: Number((await ds.query(`SELECT count(*)::int n FROM migrations`))[0].n),
    schema: brief(fp2),
    columnsRemoved: {
      'org_units.status/archivedAt': !(await has(`SELECT 1 FROM information_schema.columns WHERE table_name='org_units' AND column_name IN ('status','archivedAt')`)),
      'vehicles.orgUnitId': !(await has(`SELECT 1 FROM information_schema.columns WHERE table_name='vehicles' AND column_name='orgUnitId'`)),
      'sales_orders.branchOrgUnitId': !(await has(`SELECT 1 FROM information_schema.columns WHERE table_name='sales_orders' AND column_name='branchOrgUnitId'`)),
      'org_migration_log table': !(await has(`SELECT 1 FROM information_schema.tables WHERE table_name='org_migration_log'`)),
      'single-root index': !(await has(`SELECT 1 FROM pg_indexes WHERE indexname='IDX_org_units_single_root'`)),
    },
    parentFkDeleteRuleAfterDown: (await ds.query(`SELECT confdeltype FROM pg_constraint WHERE conrelid='org_units'::regclass AND contype='f' AND confrelid='org_units'::regclass`)).map((r: { confdeltype: string }) => r.confdeltype),
    businessRowsAfterDown: await seeded(),
    archivedUnitStillPresentButStatusLost: Number((await ds.query(`SELECT count(*)::int n FROM org_units WHERE id=$1`, [archivedId]))[0].n) === 1,
  };

  // 5. up again
  const ran2 = await ds.runMigrations({ transaction: 'each' });
  const fp3 = await fingerprint(ds);
  out.step5_upAfterDown = {
    migrationsApplied: ran2.map((m) => m.name),
    schemaFingerprintEqualsStep1: fp3.hash === fp1.hash,
    schema: brief(fp3),
    businessRowsAfterUp: await seeded(),
    dataInNewColumnsAfterRoundTrip: {
      salesOrderBranchStamp: (await ds.query(`SELECT "branchOrgUnitId" FROM sales_orders WHERE id=$1`, [so]))[0].branchOrgUnitId,
      vehicleOrgUnit: (await ds.query(`SELECT "orgUnitId" FROM vehicles WHERE id=$1`, [veh]))[0].orgUnitId,
      archivedUnitStatus: (await ds.query(`SELECT status FROM org_units WHERE id=$1`, [archivedId]))[0].status,
      orgMigrationLogTableRows: Number((await ds.query(`SELECT count(*)::int n FROM org_migration_log`))[0].n),
    },
  };
  if (fp3.hash !== fp1.hash) {
    const a = new Set(fp1.cols.map((c: unknown) => JSON.stringify(c))), b = new Set(fp3.cols.map((c: unknown) => JSON.stringify(c)));
    (out.step5_upAfterDown as Record<string, unknown>).schemaDiff = { onlyInFirst: [...a].filter((x) => !b.has(x)), onlyInAfter: [...b].filter((x) => !a.has(x)) };
  }
  await ds.destroy();

  // 6. clean-up and proof that nothing else on the cluster changed
  await admin.query(`DROP DATABASE "${DB}" WITH (FORCE)`);
  const after = (await admin.query(`SELECT datname FROM pg_database WHERE datname NOT LIKE 'template%' ORDER BY 1`)).map((r: { datname: string }) => r.datname);
  out.cleanup = { disposableDatabaseDropped: !after.includes(DB), otherDatabasesBefore: before.length, otherDatabasesAfter: after.length };
  await admin.destroy();
  out.finishedAt = new Date().toISOString();
  console.log(JSON.stringify(out, null, 2));
}
main().catch(async (e) => {
  console.error('ERR', e.message);
  try {
    const a = new DataSource({ type: 'postgres', ...base, database: 'postgres' });
    await a.initialize();
    await a.query(`DROP DATABASE IF EXISTS "${DB}" WITH (FORCE)`);
    await a.destroy();
  } catch (_) { /* best effort */ }
  process.exit(1);
});
