// C4 HTTP-level smoke on the ISOLATED browser-verification environment (disposable databases on the guarded test cluster, backend :3604 from the C4 worktree).
// It drives the REAL guards, validation pipes and approval engine through HTTP. Passwords of the bootstrap test users are the documented fixtures of tools/testing/bv-*.ts.
const { Client } = require('pg');
const BASE = 'http://127.0.0.1:3604/api';
const COMPANY = '944bb38f-a589-4318-aab3-7d2eaa75e71f';
const PW = process.env.PW;
const out = [];
const rec = (step, expected, actual) => { const ok = JSON.stringify(expected) === JSON.stringify(actual); out.push({ step, expected, actual, pass: ok }); console.log((ok ? 'PASS ' : 'FAIL ') + step + ' | expected ' + JSON.stringify(expected) + ' | actual ' + JSON.stringify(actual)); };

async function http(method, path, token, body) {
  const r = await fetch(BASE + path, { method, headers: { 'Content-Type': 'application/json', ...(token ? { Authorization: 'Bearer ' + token } : {}) }, body: body ? JSON.stringify(body) : undefined });
  let j = null; try { j = await r.json(); } catch (e) { /* no body */ }
  return { status: r.status, data: j && (j.data !== undefined ? j.data : j), raw: j };
}
async function login(email, password) {
  const l = await http('POST', '/auth/login', null, { email, password });
  const s = await http('POST', '/auth/select-company', l.data.accessToken, { companyId: COMPANY });
  return s.data.accessToken;
}

  const admin = await login('bv.admin@example.test', 'BvAdmin#2026');
  const a1 = await login('bv.approver1@example.test', 'BvApprover#2026');
